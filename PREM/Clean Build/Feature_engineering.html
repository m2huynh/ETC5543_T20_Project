db_data <- raw_data |>
  
  # date
  mutate(
    match_date = format(start_date, "%d/%m/%y")
  ) |>
  relocate(match_date, .after = season) |>
  
  
  # remove / relocate unnecessary variables
  relocate(start_date, .after = fielder_3) |>
  select(
    -penalty,
    -other_wicket_type
  ) |>
  
  
  # over
  mutate(
    over = floor(actual_delivery) + 1
  ) |>
  relocate(over, .after = actual_delivery) |>
  
  
  # rearrange important delivery variables
  relocate(
    runs_off_bat,
    extras,
    wides,
    noballs,
    byes,
    legbyes,
    wicket_type,
    .after = over
  ) |>
  
  # phase of the game
  mutate(
    Phase_over = case_when(
      over <= 6 ~ "Powerplay",
      over <= 16 ~ "Mid-Game",
      over <= 20 ~ "Death"
    )
  ) |>
  relocate(Phase_over, .after = over) |>
  
  #delivery order
  arrange(start_date, match_id, innings) |> 
   group_by(match_id,
            innings) |>
  mutate(  delivery_order = row_number() ) |>
  
  ungroup() |>
  
   # target runs
  left_join(
    raw_data_1 |>
      select(match_id, target_runs),
    by = "match_id") |>
  relocate(target_runs, .after = innings) |>
  
  # wicket on current delivery
  mutate(
    wicket_this_ball = case_when(
      is.na(wicket_type) ~ 0,
      wicket_type == "retired hurt" ~ 0,
      TRUE ~ 1) ) |>
  relocate(
    wicket_this_ball, .after = wicket_type) |>
  
  #replace NA with 0
  mutate(
    extras = ifelse(is.na(extras), 0, extras),
    wides = ifelse(is.na(wides), 0, wides),
    noballs = ifelse(is.na(noballs), 0, noballs),
    byes = ifelse(is.na(byes), 0, byes),
    legbyes = ifelse(is.na(legbyes), 0, legbyes) 
    ) |>
  
  # runs on current delivery
  mutate(runs_this_ball = runs_off_bat + extras) |>
  relocate(
    runs_this_ball, .after = wicket_this_ball) |>
  
  
  #rearrange main match variables
  relocate(
    innings,
    target_runs,
    ball,
    actual_delivery,
    over,
    Phase_over,
    wides,
    noballs,
    byes,
    legbyes,
    extras,
    wicket_this_ball,
    wicket_type,
    runs_off_bat,
    runs_this_ball,
    .after = match_date
  ) |>
  
  #cumulative match score
  group_by(match_id, innings) |>
  mutate(
    cumulative_runs = cumsum(runs_this_ball),
    cumulative_wickets = cumsum(wicket_this_ball),
    score = paste0(
      cumulative_runs,
      "/",
      cumulative_wickets
    )
  ) |>
  ungroup() |>
  
  # cumulative extras
  group_by(
    match_id,
    innings
  ) |>
  mutate(
    cumulative_extras = cumsum(extras),
    cumulative_wides = cumsum(wides),
    cumulative_noballs = cumsum(noballs),
    cumulative_byes = cumsum(byes),
    cumulative_legbyes = cumsum(legbyes) ) |>
  ungroup() |>
  
  
   # phase cumulative score
  group_by(
    match_id,
    innings,
    Phase_over
  ) |>
  mutate(
    phase_runs = cumsum(runs_this_ball),
    phase_wickets = cumsum(wicket_this_ball),
    phase_extras = cumsum(extras),
    phase_wides = cumsum(wides),
    phase_noballs = cumsum(noballs),
    phase_byes = cumsum(byes),
    phase_legbyes = cumsum(legbyes) ) |>
  ungroup() |>
  
  
  #legal delivery
  mutate(
    legal_ball = case_when(
      !is.na(wides) & wides > 0 ~ 0,
      !is.na(noballs) & noballs > 0 ~ 0,
      TRUE ~ 1
    )
  ) |>
  
  #cumulative legal deliveries
  group_by(
    match_id,
    innings
  ) |>
  mutate(
    cumulative_legal_balls = cumsum(legal_ball)
  ) |>
  ungroup() |>
  
  # runs, wickets before current ball
  mutate(
    runs_before_ball = cumulative_runs - runs_this_ball,
    wickets_before_ball = cumulative_wickets - wicket_this_ball
  ) |>
  relocate(
    runs_before_ball,
    .after = runs_this_ball
  ) |>
  relocate(
    wickets_before_ball,
    .after = wicket_this_ball
  ) |>

  
  #current team run rate
  mutate(
    team_run_rate = case_when(
      cumulative_legal_balls > 0 ~
        (cumulative_runs / cumulative_legal_balls) * 6,
      TRUE ~ NA_real_
    )
  )

pressure_data <- db_data |>
  
  # only second innings = chasing team and removing interrupted matches - rain/retired hurt batsmen
  filter(
    innings == 2,
    !is.na(target_runs),
    !match_id %in% shortened_matches$match_id
  ) |>
  
  # add match winner
  left_join(
    raw_data_1 |>
      select(
        match_id,
        winner
      ),
    by = "match_id"
  ) |>
  
  # create match state BEFORE the current delivery
  mutate(
    
    legal_balls_before_ball =
      cumulative_legal_balls - legal_ball,
    
    balls_remaining =
      120 - legal_balls_before_ball,
    
    runs_required =
      pmax(
        target_runs - runs_before_ball,
        0
      ),
    
    wickets_in_hand =
      10 - wickets_before_ball,
    
    required_run_rate =
      case_when(
        balls_remaining > 0 ~
          (runs_required / balls_remaining) * 6,
        TRUE ~ NA_real_
      ),
    
    # final outcome of the chase
    chase_win =
      case_when(
        is.na(winner) ~ NA_integer_,
        batting_team == winner ~ 1L,
        TRUE ~ 0L
      )
  ) |>
  
  # keep only variables required for the model
  select(
    match_id,
    season,
    required_run_rate,
    wickets_in_hand,
    balls_remaining,
    chase_win
  ) |>
  
  # remove unusable observations
  filter(
    !is.na(required_run_rate),
    !is.na(wickets_in_hand),
    !is.na(balls_remaining),
    !is.na(chase_win),
    balls_remaining > 0
  )

batter_pressure_data <- db_data |>
  
  # only valid chasing innings
  filter(
    innings == 2,
    !is.na(target_runs),
    !match_id %in% shortened_matches$match_id,
    !is.na(striker)
  ) |>
  
  # recreate match state before each delivery
  mutate(
    
    legal_balls_before_ball =
      cumulative_legal_balls - legal_ball,
    
    balls_remaining =
      120 - legal_balls_before_ball,
    
    runs_required =
      pmax(
        target_runs - runs_before_ball,
        0
      ),
    
    wickets_in_hand =
      10 - wickets_before_ball,
    
    required_run_rate =
      case_when(
        balls_remaining > 0 ~
          (runs_required / balls_remaining) * 6,
        TRUE ~ NA_real_
      ),
    
    # wides do not count as balls faced
    batter_ball =
      case_when(
        wides > 0 ~ 0,
        TRUE ~ 1
      )
  ) |>
  
  # remove unusable match states
  filter(
    !is.na(required_run_rate),
    !is.na(wickets_in_hand),
    !is.na(balls_remaining),
    balls_remaining > 0
  )
